function verifyUserAuthorization(role: string[]) {
  return (
    request: Request,
    response: Response,
    next: NextFunction,
  ) => {
    if (!request.user || !role.includes(request.user.role)) {
      throw new AppError("Acesso negado", 403)
    }

    return next()
  }
}

@@ RefundPay/read-token.ts
function readToken(request: Request) {
  const authHeader = request.headers.authorization

  if (!authHeader) {
    throw new AppError("JWT token not found", 401)
  }

  const [scheme, token] = authHeader.split(" ")

  if (scheme !== "Bearer" || !token) {
    throw new AppError("Invalid JWT token", 401)
  }

  try {
    return verify(token, authConfig.jwt.secret, {
      algorithms: ["HS256"],
    }) as TokenPayload
  } catch {
    throw new AppError("Invalid JWT token", 401)
  }
}
%% Lê o token de login que o front manda em cada requisição do RefundPay e confere se ele é válido.
1: Recebe o pedido que chegou no Express.
2: Pega o cabeçalho Authorization, que vem no formato "Bearer <token>".
4-6: Sem cabeçalho, ninguém está logado: erro 401 (não autenticado).
8: Separa o texto no espaço e pega as duas partes pela posição (desestruturação de array).
10-12: Se a primeira parte não for "Bearer" ou se faltar o token, também é 401.
14-17: O verify confere a assinatura com o segredo do servidor e devolve os dados de dentro do token. Fixar o algoritmo HS256 impede que alguém mande um token assinado de outro jeito.
18-20: Se o token estiver vencido ou adulterado, o verify lança erro, que vira um 401 com a mesma mensagem.

async function review(
  id: string,
  reviewer: Reviewer,
  status: "approved" | "rejected",
) {
  const { count } = await prisma.refunds.updateMany({
    where: {
      id,
      organizationId: reviewer.organizationId,
      status: "pending",
    },
    data: {
      status,
      reviewedById: reviewer.id,
      reviewedAt: new Date(),
    },
  })

  if (count === 0) {
    throw new AppError("Essa solicitação já foi analisada", 409)
  }
}

7-11: O filtro: aquele pedido, da empresa de quem está analisando e ainda pendente. É por isso que uma empresa nunca mexe no pedido de outra.
12-16: O que muda: o status novo, quem analisou e quando.

@@ RefundPay/file-type.ts
const SIGNATURES = [
  { mime: "image/jpeg", bytes: [0xff, 0xd8, 0xff] },
  { mime: "image/png", bytes: [0x89, 0x50, 0x4e, 0x47] },
  { mime: "application/pdf", bytes: [0x25, 0x50, 0x44, 0x46] },
]

export async function detectReceiptType(filePath: string) {
  const handle = await fs.promises.open(filePath, "r")

  try {
    const header = Buffer.alloc(8)
    await handle.read(header, 0, header.length, 0)

    return SIGNATURES.find((type) =>
      type.bytes.every((byte, i) => header[i] === byte),
    ) ?? null
  } finally {
    await handle.close()
  }
}
%% Descobre se o comprovante enviado ao RefundPay é mesmo JPG, PNG ou PDF olhando os primeiros bytes do arquivo, porque o tipo que o navegador informa pode ser falsificado.
1-5: A "assinatura" de cada formato: todo arquivo desse tipo começa com esses bytes. Os números com 0x estão em hexadecimal, que é só outro jeito de escrever números. O 0x25 0x50 0x44 0x46 é o texto "%PDF".
7: Recebe o caminho do arquivo salvo no servidor.
8: Abre o arquivo só para leitura ("r").
11-12: Cria um espaço de 8 bytes (Buffer) e lê o começo do arquivo para dentro dele.
14-16: Procura o formato em que todos os bytes da assinatura batem com o começo do arquivo. Se nenhum bater, o ?? devolve null e o envio é recusado.
17-19: O finally roda sempre, dando certo ou errado, e garante que o arquivo seja fechado.

export function updateStreak(won: boolean): UserStats {
  const stats = loadStats()
  if (won) {
    stats.streak += 1
    if (stats.streak > stats.bestStreak) {
      stats.bestStreak = stats.streak

  } else {
    stats.streak = 0
  }
  saveStats(stats)
  return stats
}

export function speakWord(text: string, rate = 0.85) {
  return new Promise<void>((resolve) => {
    window.speechSynthesis.cancel()

    const utterance = new SpeechSynthesisUtterance(
      text.trim().toLowerCase(),
    )
    utterance.lang = "en-US"
    utterance.rate = rate

    utterance.onend = () => resolve()
    utterance.onerror = () => resolve()

    window.speechSynthesis.speak(utterance)
  })
}

@@ aprendendo-vue/useProspeccoes.js
export function useProspeccoes() {
  const prospeccoes = ref(carregar())

  watch(
    prospeccoes,
    (lista) => {
      localStorage.setItem(CHAVE, JSON.stringify(lista))
    },
    { deep: true },
  )

  function mudarEtapa(id, etapa) {
    const prospeccao = prospeccoes.value.find((p) => p.id === id)
    if (prospeccao) prospeccao.etapa = etapa
  }

  function remover(id) {
    prospeccoes.value = prospeccoes.value.filter(
      (p) => p.id !== id,
    )
  }

  return { prospeccoes, mudarEtapa, remover }
}
%% O composable do meu funil de prospecção no aprendendo-vue (versão resumida): guarda a lista, salva sozinho e muda a etapa de cada negócio.
1: Um composable é o custom hook do Vue: uma função que guarda estado e devolve o que o componente precisa.
2: A lista começa com o que estava salvo, porque a função carregar lê o localStorage.
4-10: Sempre que a lista mudar, salva de novo. É como um useEffect com a lista nas dependências, e o deep: true pega até mudança dentro de um item.
12-15: Acha a prospecção pelo id e troca a etapa direto no objeto. No React você teria que criar um array novo e chamar o set; no Vue, alterar já basta.
17-21: Para remover, troca a lista inteira por uma cópia filtrada sem aquele id.
23: Devolve o estado e as funções. No componente fica assim: const { prospeccoes, remover } = useProspeccoes().
